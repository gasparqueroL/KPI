```python
def aging_cobros(db: Session) -> dict[str, Any]:
    """Aging de saldos pendientes por cliente, agrupado en buckets temporales.

    Buckets calculados sobre la fecha de la venta pendiente (Venta.fecha):
    - bucket_0_30: ventas pendientes con <= 30 días desde emisión
    - bucket_31_60: 31-60 días
    - bucket_61_90: 61-90 días
    - bucket_90_mas: > 90 días
    """
    from collections import defaultdict
    from datetime import date, timedelta

    hoy = date.today()
    cortes = {
        "bucket_0_30": (None, 30),
        "bucket_31_60": (30, 60),
        "bucket_61_90": (60, 90),
        "bucket_90_mas": (90, None),
    }

    # Cargo total por cliente (igual que en clientes_con_cuenta)
    pagado_por_cliente = dict(
        db.query(
            MovimientoCaja.id_cliente_relacionado,
            func.coalesce(func.sum(MovimientoCaja.monto), 0),
        ).filter(
            MovimientoCaja.id_cliente_relacionado.isnot(None),
            MovimientoCaja.caja_destino.isnot(None),
        ).group_by(MovimientoCaja.id_cliente_relacionado).all()
    )

    # Ventas pendientes por cliente con fecha
    ventas = db.query(Venta).filter(
        Venta.id_cliente.isnot(None),
        or_(
            Venta.es_cuenta_corriente == True,
            (
                (func.coalesce(Venta.monto_pago1, 0) + func.coalesce(Venta.monto_pago2, 0)) == 0
            ) & (Venta.total > 0),
        ),
    ).order_by(Venta.fecha).all()

    # Para cada cliente, repartir las ventas en buckets y restar pagos en orden FIFO
    from collections import defaultdict
    por_cliente: dict = defaultdict(lambda: {"nombre": None, "buckets": defaultdict(Decimal), "total": Decimal(0)})

    for v in ventas:
        # Fecha base para el aging — la fecha de la venta
        v_fecha = v.fecha.date() if hasattr(v.fecha, "date") else v.fecha
        dias = (hoy - v_fecha).days
        bucket = (
            "bucket_0_30" if dias <= 30
            else "bucket_31_60" if dias <= 60
            else "bucket_61_90" if dias <= 90
            else "bucket_90_mas"
        )
        por_cliente[v.id_cliente]["nombre"] = v.cliente or por_cliente[v.id_cliente]["nombre"]
        por_cliente[v.id_cliente]["buckets"][bucket] += Decimal(str(v.total))
        por_cliente[v.id_cliente]["total"] += Decimal(str(v.total))

    # Restar pagos por cliente — se aplican primero a buckets más viejos (FIFO inverso)
    # para representar "el cliente paga primero lo más viejo"
    orden_buckets = ["bucket_90_mas", "bucket_61_90", "bucket_31_60", "bucket_0_30"]
    for cli_id, info in por_cliente.items():
        pagado = Decimal(str(pagado_por_cliente.get(cli_id, 0)))
        restante = pagado
        for b in orden_buckets:
            if restante <= 0:
                break
            si = info["buckets"][b]
            if si > 0:
                aplicado = min(si, restante)
                info["buckets"][b] -= aplicado
                restante -= aplicado
        info["total"] -= pagado

    # Convertir a output
    items = []
    totales = {b: Decimal(0) for b in cortes}
    total_global = Decimal(0)
    for cli_id, info in por_cliente.items():
        if info["total"] <= 0:
            continue
        item = {
            "id_cliente": cli_id,
            "cliente": info["nombre"],
            "total": _to_float(info["total"]),
            **{b: _to_float(info["buckets"][b]) for b in cortes},
        }
        items.append(item)
        for b in cortes:
            totales[b] += info["buckets"][b]
        total_global += info["total"]

    items.sort(key=lambda r: r["bucket_90_mas"], reverse=True)

    return {
        "totales": {**{b: _to_float(v) for b, v in totales.items()}, "total": _to_float(total_global)},
        "items": items,
    }


def dso(db: Session, dias_ventana: int = 90) -> dict[str, Any]:
    """Days Sales Outstanding: días promedio que tardan los clientes en pagar.

    Fórmula: (saldo_pendiente / ventas_periodo) * dias_ventana
    Saldo pendiente = sum(ventas pendientes con fecha en últimos N días)
    Ventas periodo = sum(ventas totales en últimos N días)
    """
    from datetime import date, datetime, time, timedelta

    hoy = date.today()
    desde = hoy - timedelta(days=dias_ventana)

    desde_dt = datetime.combine(desde, time.min)

    ventas_total = db.query(func.coalesce(func.sum(Venta.total), 0)).filter(
        Venta.fecha >= desde_dt
    ).scalar()
    ventas_total = Decimal(str(ventas_total or 0))

    # Pendiente = ventas con sin_cobro o cta_cte en mismo período
    pendiente = db.query(func.coalesce(func.sum(Venta.total), 0)).filter(
        Venta.fecha >= desde_dt,
        or_(
            Venta.es_cuenta_corriente == True,
            (
                (func.coalesce(Venta.monto_pago1, 0) + func.coalesce(Venta.monto_pago2, 0)) == 0
            ) & (Venta.total > 0),
        ),
    ).scalar()
    pendiente = Decimal(str(pendiente or 0))

    if ventas_total > 0:
        dso_val = (pendiente / ventas_total) * dias_ventana
    else:
        dso_val = Decimal(0)

    return {
        "dias_ventana": dias_ventana,
        "ventas_periodo": _to_float(ventas_total),
        "saldo_pendiente_periodo": _to_float(pendiente),
        "dso_dias": round(float(dso_val), 1),
    }
```

```python
def test_aging_distribuye_ventas_en_buckets(db):
    """Ventas de fechas distintas caen en buckets distintos."""
    from datetime import date, timedelta
    hoy = date.today()
    db.add(Caja(nombre_normalizado="cl", nombre_display="CAJA LOCAL", tipo="operativa"))
    # Venta de hace 10 días -> bucket 0-30
    db.add(Venta(
        id_pedido="p1", id_venta="v1",
        fecha=datetime.combine(hoy - timedelta(days=10), datetime.min.time()),
        id_cliente=1, cliente="A", total=Decimal("1000"),
        es_cuenta_corriente=True,
    ))
    # Venta de hace 45 días -> bucket 31-60
    db.add(Venta(
        id_pedido="p2", id_venta="v2",
        fecha=datetime.combine(hoy - timedelta(days=45), datetime.min.time()),
        id_cliente=1, cliente="A", total=Decimal("2000"),
        es_cuenta_corriente=True,
    ))
    # Venta de hace 100 días -> bucket 90+
    db.add(Venta(
        id_pedido="p3", id_venta="v3",
```
